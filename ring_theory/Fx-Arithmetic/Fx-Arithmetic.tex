\section{Problem 1}
Write out the addition and multiplication tables for the congruence class rings $F[x]/p(x)$. In each case, is $F[x]/p(x)$ a field?

\subsection{Let $F = \mathbb{Z}_3$ and $p(x) = x^2 + 1$}

\begin{table}[h]
    \centering
    \begin{tabular}{c|ccc}
        + & 0 & 1 & x \\
        \hline
        0 & 0 & 1 & x \\
        1 & 1 & 2 & x+1 \\
        x & x & x+1 & 2x \\
    \end{tabular}
    \caption{Addition table for \(\mathbb{Z}_3 / (x^2+1)\)}
\end{table}


\subsection{Let $F = \mathbb{Z}_2$ and $p(x) = x^2 + 1$}

        1 & 1 & 0 & x+1 \\
        x & x & x+1 & 0 \\
    \end{tabular}
    \caption{Addition table for \(\mathbb{Z}_2 / (x^2+1)\)}
