\section*{Problem 1}
\subsection*{If $F$ is a field, $R$ is a nonzero ring, and $f: F \to R$ a surjective homomorphism, 
prove that $f$ is an isomorphism.}
\begin{proof}
    The First Isomorphism Theorem states that if $f: F \to R$ is a surjective homomorphism, then $F/ker(f) \cong Im(f)$.
    Since $F$ is a field, then its only ideals are $\{0\}$ and $F$ itself.
    The kernel of $f$ is either $\{0\}$ or $F$.
    If the kernel of $f$ is $\{0\}$, then $f$ is injective because $F / ker(f) = F$ and thus the isomorphism
    gives $F \cong R$.
    If the kernel of $f$ is $F$, then $F/ ker(f) = \{0\}$ and thus the isomorphism gives $0 \cong R$.
    This is a contradiction because $0$ is not a ring and $R$ is a nonzero ring.
    Therefore, the kernel of $f$ must be $\{0\}$ and $f$ is an isomorphism.
\end{proof}

\section*{Problem 2}
\subsection*{The function $\phi: \mathbb{R}[x] \to \mathbb{R}$ given by $\phi(f(x)) = f(2)$ is a homomorphism of rings. Find
the kernel of $\phi$.}
\begin{proof}
    The kernel of $\phi$ is the set of polynomials $f(x) \in \mathbb{R}[x]$ such that $\phi(f(x)) = 0$.
    This is equivalent to the set of polynomials $f(x) \in \mathbb{R}[x]$ such that $f(2) = 0$.
    Therefore, the kernel of $\phi$ is $(x-2)$.
\end{proof}

\section*{Problem 3}
$\mathbb{Z}[\sqrt{2}]$ is a ring. Let $f: \mathbb{Z}[\sqrt{2}] \to \mathbb{Z}[\sqrt{2}]$ be defined by $f(a+b\sqrt{2}) = a-b\sqrt{2}$.
\subsection*{Show that $f$ is a surjective homomorphism of rings.}
\begin{proof}
    We first show that $f$ is a homomorphism of rings.
    For any two elements $x = a+b\sqrt{2}, y = c+d\sqrt{2} \in \mathbb{Z}[\sqrt{2}]$, we have
    \[
        f(x+y) = f((a+c) + (b+d)\sqrt{2}) = (a+c) - (b+d)\sqrt{2} = a-b\sqrt{2} + c-d\sqrt{2} = f(x) + f(y).
    \]
    And for any two elements $a, b \in \mathbb{Z}[\sqrt{2}]$, we have
    \[
        f(xy) = f((a+b\sqrt{2})(c+d\sqrt{2})) = f((ac+2bd) + (ad+bc)\sqrt{2}) = (ac+2bd) - (ad+bc)\sqrt{2} = f(x)f(y).
    \]
    Therefore, $f$ is a homomorphism of rings.
    We now show that $f$ is surjective.
    For any given element $a+b\sqrt{2} \in \mathbb{Z}[\sqrt{2}]$, we can find some $x = a-b\sqrt{2} \in \mathbb{Z}[\sqrt{2}]$ such that $f(x) = a-b\sqrt{2}$.
    Therefore, $f$ is surjective.
\end{proof}

We know that the kernel of $f$ is $(x-2)$.
Since $(x-2)$ is a principal ideal, then it is a maximal ideal.
Therefore, by Theorem 6.11, we know that $f$ is injective.
Since $f$ is both injective and surjective, then $f$ is an isomorphism.
\end{proof}

\section*{Problem 4}
Suppose $I$ and $J$ are ideals in a ring $R$ and let $f: R \to R/I \times R/J$ be the function defined by $f(a) = (a+I, a+J)$.
\subsection*{Prove that $f$ is a homomorphism of rings.}
\begin{proof}
    We first show that $f$ is a homomorphism of rings.
    For any two elements $a, b \in R$, we have
    \[
        f(a+b) = (a+b+I, a+b+J) = (a+I, b+J) = f(a) + f(b).
    \]
    And for any two elements $a, b \in R$, we have
    \[
        f(ab) = (ab+I, ab+J) = (a+I, b+J) = f(a)f(b).
    \]
    Finally if $1 \in R$, then $f(1) = (1+I, 1+J)$ Since $1+I \in R/I$ and $1+J \in R/J$, then $f(1) \in R/I \times R/J$.
    Therefore, $f$ is a homomorphism of rings.
\end{proof}

\subsection*{Is $f$ surjective?}
\begin{proof}
    We now show that $f$ is surjective.
    For any given coset $a+I \in R/I$ and $a+J \in R/J$, we can find some $a \in R$ such that $f(a) = (a+I, a+J)$.
    We call $f$ the natural homomorphism from $R$ to $R/I \times R/J$. Thus $f$ is indeed surjective.
\end{proof}

\subsection*{What is the kernel of $f$?}
\begin{proof}
    The kernel of $f$ is the set of elements $a \in R$ such that $f(a) = (0+I, 0+J)$.
    This is equivalent to the set of elements $a \in R$ such that $a+I = 0+I$ and $a+J = 0+J$.
    Therefore, the kernel of $f$ is $I \cap J$.
\end{proof}

\section*{Problem 5}
\subsection*{Use the First Isomorphism Theorem to show that $\mathbb{Z}_{20}/(5) \cong \mathbb{Z}_5$.}
\begin{proof}
    Let $f: \mathbb{Z}_{20} \to \mathbb{Z}_5$ be defined by $f[x]_{20} = [x]_5$, where $[x]_n$ denotes an
    element of $\mathbb{Z}_n$.
    For any two elements $[x]_{20}, [y]_{20} \in \mathbb{Z}_{20}$, we have
    \[
        f([x]_{20} + [y]_{20}) = f([x+y]_{20}) = [x+y]_5 = [x]_5 + [y]_5 = f([x]_{20}) + f([y]_{20}).
    \]
    Thus $f$ is a well-defined homomorphism of rings.
    We now show that $f$ is surjective. The kernel of $f$ is the set of elements $[x]_{20} \in \mathbb{Z}_{20}$
    such that $f([x]_{20}) = [0]_5$.
    This is equivalent of saying the kernel of $f$ consists of all multiples of $5 \in \mathbb{Z}_{20}$.
    Therefore, the kernel of $f$ is $(5) = \{[0]_{20}, [5]_{20}, [10]_{20}, [15]_{20}\}$.
    By the First Isomorphism Theorem, we have
    \[
        \mathbb{Z}_{20}/ker(f) \cong Im(f).
    \]
    Since the kernel of $f$ is $(5)$, we have
    \[
        \mathbb{Z}_{20}/(5) \cong \mathbb{Z}_5.
    \]
\end{proof}

\section*{Problem 7}
\subsection*{Prove that a nonzero integer $p$ is prime if and only if the ideal $(p)$ is
maximal in $\mathbb{Z}$.}
\begin{proof}
    $(\Rightarrow)$ Suppose that $p$ is prime. Then $(p)$ is a prime ideal. From Theorem 6.14 in Hungerford, we know that
    (p) is a prime ideal if and only if $\mathbb{Z}$ is an integral domain.
    Since $\mathbb{Z}$ is an integral domain, then $(p)$ is a prime ideal. Assume there exists an ideal $I$ such that $(p) \subseteq I \subseteq \mathbb{Z}$.
    If $1 \in I$, then $I = \mathbb{Z}$ since any ideal containing $1$ must be the whole ring. If $1 \notin I$, then $I \subset (p)$. Since $p$ is prime, then
    it must be that for any element $a \in I$ must satisfy $a = kp$ for some $k \in \mathbb{Z}$.
    Therefore either $I = (p)$ or $I = \mathbb{Z}$ and we conclude that $(p)$ is maximal.
    
    $(\Leftarrow)$ Suppose that $(p)$ is a maximal ideal. Then by Theorem 6.15 in Hungerford, we should have
    it that $\mathbb{Z}/(p)$ is a field. Since $\mathbb{Z}/(p)$ is isomorphic to $\mathbb{Z}_p$, then $\mathbb{Z}/(p)$ is a field
    and $p$ is prime.
\end{proof}

\subsection*{Let $F$ be a field and $p(x) \in F[x]$. Prove that $p(x)$ is irreducible
if and only if the ideal $(p(x))$ is maximal in $F[x]$.}
\begin{proof}
    $(\Rightarrow)$ Suppose that $p(x)$ is irreducible and there exists an ideal $I$ such that $(p(x)) \subseteq I \subseteq F[x]$.
    We now show that $I = (p(x))$ or $I = F[x]$. In the case where $I = F[x]$, there is nothing more to show.
    In the case where $I \neq F[x]$, then we consider some arbitrary polynomial$f(x) \in I$.
    Since $p(x)$ is irreducible, there are two cases:
    \begin{enumerate}
        \item \textbf{Case 1:} If $f(x)$ is a multiple of $p(x)$, i.e. $f(x) = p(x)g(x)$ for some $g(x) \in F[x]$.
        Then $f(x) \in (p(x))$ and we have $I \subseteq (p(x))$.
        \item \textbf{Case 2:} If $f(x)$ is coprime to $p(x)$, then there exists some $a(x), b(x) \in F[x]$ such that $a(x)p(x) + b(x)f(x) = 1$.
        Since $a(x)p(x) \in (p(x))$ and $b(x)f(x) \in I$, then $1 \in I$ and $I = F[x]$.
    \end{enumerate}
    We have shown that $I = (p(x))$ or $I = F[x]$ and we conclude that $(p(x))$ is maximal.

    $(\Leftarrow)$ Suppose that $(p(x))$ is maximal. For the sake of contradiction we also assume that $p(x)$
    is reducible and can be expressed as $p(x) = f(x)g(x)$ where $f(x), g(x)$ are non-unit polynomials. That means in the quotient ring $F[x]/(p(x))$ we have
    [p(x)] = [f(x)][g(x)] = [0]. By Theorem 6.15 in Hungerford, we know that $F[x]/(p(x))$ is a field. 
    Since $F[x]/(p(x))$ is a field, then it has no zero divisors. This is a contradiction and thus $p(x)$ must be irreducible.
\end{proof}

\section*{Problem 8}
\subsection*{Show that the principal ideal $(x-1)$ in $\mathbb{Z}[x]$ is
prime but not maximal.}
\begin{proof}
    We first show that $(x-1)$ is prime. Suppose that $f(x), g(x) \in \mathbb{Z}[x]$. Then $f(x)g(x) \in (x-1)$ if and only if $x-1$ divides $f(x)g(x)$.
    This means there is some polynomial $h(x) \in \mathbb{Z}[x]$ such that $f(x)g(x) = (x-1)h(x)$.
    We evaluate $f(1)g(1) = (1-1)h(1) = 0$.
    Since $\mathbb{Z}$ is an integral domain, then $f(1) = 0$ or $g(1) = 0$.
    Therefore, $x-1$ divides $f(x)$ or $x-1$ divides $g(x)$.
    Therefore, $(x-1)$ is prime.

    We now show that $(x-1)$ is not maximal.
    Consider the map $\phi: \mathbb{Z}[x] \to \mathbb{Z}$ given by $\phi(f(x)) = f(1)$.
    The kernel of $\phi$ is the set of polynomials $f(x) \in \mathbb{Z}[x]$ such that $f(1) = 0$.
    This is equivalent to the set of polynomials $f(x) \in \mathbb{Z}[x]$ such that $x-1$ divides $f(x)$.
    Therefore, the kernel of $\phi$ is $(x-1)$.
    By the First Isomorphism Theorem, we have
    \[
        \mathbb{Z}[x]/(x-1) \cong \mathbb{Z}.
    \]
    We know that $\mathbb{Z}$ is not a field, therefore the ideal $(x-1)$ is prime but not maximal.
\end{proof}

\section*{Problem 9}
\subsection*{Prove that $R = {a+bi | a,b \in \mathbb{Z}}$ is a subring
of $\mathbb{C}$ and that $M = {a+bi | 3\mid a \text{ and } 3\mid b}$ is a maximal
ideal of $R$.}
\begin{proof}
    We first show that $R$ is a subring of $\mathbb{C}$.
    We begin by showing that $R$ is closed under addition and multiplication.
    \begin{align*}
        (a+bi) + (c+di) &= (a+c) + (b+d)i \\
        (a+bi)(c+di) &= (ac-bd) + (ad+bc)i
    \end{align*}
    Since $a,b,c,d \in \mathbb{Z}$, then $a+c, b+d, ac-bd, ad+bc \in \mathbb{Z}$.
    Therefore, $R$ is closed under addition and multiplication.
    Also since $1 = 1 + 0i \in R$, then $R$ contains the multiplicative identity.
    Therefore, $R$ is a subring of $\mathbb{C}$.

    We now show that $M$ is a maximal ideal of $R$.
    First we show that $M$ is closed under subtraction and has the absorption property.
    If $z_1 = a+bi$ and $z_2 = c+di$ where $3 \mid a, 3 \mid b, 3 \mid c, 3 \mid d$, then
    \[
        z_1 - z_2 = (a+bi) - (c+di) = (a-c) + (b-d)i
    \] 
    Since $3 \mid a-c$ and $3 \mid b-d$, then $z_1 - z_2 \in M$.
    Next we show that $M$ has the absorption property.
    If $z_1 = a+bi \in R$ and $z_2 = c+di \in M$ where $3 \mid a, 3 \mid b, 3 \mid c, 3 \mid d$, then
    \[
        z_1z_2 = (a+bi)(c+di) = (ac-bd) + (ad+bc)i
    \] 
    Since $3 \mid ac-bd$ and $3 \mid ad+bc$, then $z_1z_2 \in M$.
    Therefore, $M$ is an ideal of $R$.

    We now show that $M$ is maximal.
    By Theorem 6.15 in Hungerford, we know that an ideal $M$ of $R$ is maximal if and only if the quotient ring $R/M$ is a field. 
    The elements of $R/M$ are equivalence classes of $R$ modulo $M$, meaning that elements $a+bi$ and $c+di$ are equivalent if and only if 
    $(a-c) + (b-d)i \in M$.
    This is equivalent to saying that $3 \mid a-c$ and $3 \mid b-d$.
    Therefore, the equivalence classes of $R/M$ are

        \{0+0i\}, \{1+0i\}, \{2+0i\}, \{0+1i\}, \{1+1i\}, \{2+1i\}, \{0+2i\}, \{1+2i\}, \{2+2i\}
    \]
    Since $R/M$ is the quotient ring of $R$ by $M$, it inherits the ring structure where both addition and multiplication in $R/M$ are closed
    and the identity elements for both operations exists. 
    We now construct the multiplication table for $R/M$.

\subsection*{Use norms to explain why in $\mathbb{Z}[i] \text{, } 2+5i$ is not a factor of
$1+6i$.}
\begin{proof}
    The norm of $2+5i$ is $2^2 + 5^2 = 29$. The norm of $1+6i$ is $1^2 + 6^2 = 37$.
    Since $29$ does not divide $37$, then $2+5i$ is not a factor of $1+6i$.
\end{proof}
